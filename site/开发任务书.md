### JSON → JS 转换脚本

```python
import json, glob, os
BASE = '/path/to/ZZZ'
DATA = f'{BASE}/data/agents'
items = [json.load(open(fp)) for fp in glob.glob(f'{DATA}/*.json')]
js = 'window.__AGENTS__ = ' + json.dumps(items, separators=(',',':')) + ';'
with open(f'{BASE}/agents/agents-data.js', 'w') as f:
    f.write(f'// {len(items)} agents\n{js}')
print(f'Saved {len(items)} agents')
```
